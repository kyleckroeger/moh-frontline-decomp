# Gameplay for the first minutes of a level (start the cursor at update 3300; the
# player has control from about update 2800 in the levels tried).
at 3300
stick 0 72 200            # walk forward
cstick 72 0 30            # turn right
trigger R 10              # fire
wait 40
trigger R 10
wait 40
stick 0 72 200 R          # advance while firing
press X 10                # jump
wait 60
cstick -72 0 60           # turn left
stick -72 0 90            # strafe left
stick 72 0 90             # strafe right
press UP 10               # next weapon
wait 60
trigger R 10              # fire or throw it
wait 90
press UP 10
wait 60
trigger R 10
wait 90
press UP 10
wait 60
trigger R 10
wait 120
press DOWN 10             # back
wait 60
press DOWN 10
wait 60
press DOWN 10
wait 60
cstick 0 40 20            # look up
cstick 0 -40 20           # and down
press B 10
wait 60
press Y 10
wait 60
press L 10
wait 60
press Z 10
wait 60
press A 10
wait 60
stick 0 72 400 R          # advance while firing
cstick 72 0 45
stick 0 72 300
trigger R 20
wait 40
stick 72 72 200 R
cstick -72 0 60
stick 0 72 400 R
stick 0 -72 200
trigger R 40
wait 300
